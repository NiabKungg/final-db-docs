ตรวจว่ามีตาราง/Sequence/Procedure ของเวอร์ชันเก่าหรืองานอื่น ตกค้างอยู่ใน user หรือไม่ — เพราะ Script ตรวจสอบของอาจารย์ (USER_TABLES) แสดงทุกตารางที่ user เป็นเจ้าของ ถ้ามีขยะตกค้างจะโชว์ปนมาในผลตรวจทันที

อธิบายการทำงาน

  • ขั้นตอนที่ 1: SELECT ก่อนเสมอ — SELECT object_type, object_name FROM user_objects WHERE object_type IN ('TABLE','SEQUENCE','PROCEDURE','VIEW') — ดูก่อนว่ามีอะไรอยู่จริง ห้ามลบมั่ว
  • ขั้นตอนที่ 2: DROP เฉพาะของเราเท่านั้น — รายการ DROP ในไฟล์ comment ไว้ แกะใช้เฉพาะชื่อที่เจอจริงในผลตรวจ (ตารางเก่า เช่น USER_TYPE, VEHICLE, BELL_REQUEST, IOT_BOX, ALERT_STATISTICS ฯลฯ)
  • สิ่งที่ห้ามแตะ — ตาราง BA (ADMIN, DEPARTMENT, LOG ฯลฯ) และของงานคอร์สอื่นใน user เดียวกัน (SEQ_TUTOR, SEQ_STUDENT, procedure ตระกูล ADD_*/DIS_*)
ผลรันจริง 11 พ.ย. 2026: พบของเก่าของ SE ตกค้างจริง 5 procedure (SP_CANCEL_BELL, SP_COMPLETE_ALIGHT, SP_LOG_NOTIFICATION, SP_PRESS_BELL, SP_SEND_DIGITAL_SIGNAL — ล้วน INVALID เพราะอ้างตารางเก่าที่ถูกลบไปแล้ว) → DROP ทิ้งสำเร็จทั้งหมด และยืนยันว่าตารางเก่า (BELL_REQUEST, ALERT_STATISTICS ฯลฯ) ถูกเคลียร์ไปก่อนหน้าแล้ว เหลือแค่ procedure เก่า

จุดประสงค์

แปลง ER Diagram 10 เอนทิตี (Strong 9 + Weak 1) เป็นตาราง Oracle พร้อม PK/FK/UNIQUE/DEFAULT ครบตามความสัมพันธ์ 8 จุดใน ER

อธิบายการทำงานทีละตาราง

  1. User_Types — ตาราง lookup ประเภทผู้ใช้ (Admin / พนักงานขับรถ / ผู้รับบริการ) สร้างก่อนเพราะ Users จะชี้มา
  2. Users — ผู้ใช้งานระบบ; Composite Attribute "ชื่อ" ใน ER แตกเป็น first_name + last_name; FK fk_users_usertype; DEFAULT 'Active' ให้ status
  3. Routes — สายเดินรถ (แม่ของรถและรายการจุดจอด)
  4. Bus_Stops — ป้ายจอด; พิกัดจาก ER แยกเป็น latitude/longitude สองคอลัมน์ NUMBER(10,6)
  5. Route_Stops — Weak Entity จาก ER แตกความสัมพันธ์ M:N ระหว่างสายกับป้าย — PK รวม (route_id, seq): route_id = PK เจ้าของ, seq = Partial Key (ลำดับป้ายในสาย) + FK สองเส้น (fk_rs_route, fk_rs_stop)
  6. IoT_Boxes — กล่อง IoT พร้อมพิกัด GPS ที่จะถูกอัปเดตสดผ่าน Procedure
  7. Vehicles — รถ; FK 3 เส้น (สาย / IoT / คนขับ); iot_id UNIQUE ทำให้ 1 รถติดกล่องเดียว (ER เขียน M:N แต่การใช้จริง 1:1); passenger_count DEFAULT 0 = Derived Attribute เก็บเป็นค่า cache
  8. Emergency_Buttons — ปุ่มฉุกเฉินกายภาพบนรถ (1 คันหลายปุ่ม)
  9. Digital_Bells — Transaction table หลักของระบบ: ทุกครั้งที่ผู้โดยสารกดกริ่งขอลงรถจะเกิด 1 แถว (FK user + vehicle)
  10. Alert_Logs — บันทึกเหตุฉุกเฉินจากการกดปุ่ม; คอลัมน์เวลาชื่อ alert_time ไม่ใช่ timestamp (อ่านเหตุผลในกล่องเตือนด้านล่าง)
บทเรียนจริงจาก deploy 11/11 — ห้ามตั้งชื่อคอลัมน์ว่า timestamp: ตอนแรกคอลัมน์นี้ชื่อ "timestamp" ใช้ใน SQL ปกติได้ผลทุกอย่าง (สร้างตาราง/INSERT ผ่านหมด) แต่พอ compile sp_report_emergency_alert ที่ INSERT ลงตารางนี้ เกิด PLS-320 (การประกาศประเภท expression ไม่สมบูรณ์) — เพราะ "timestamp" ชนกับชื่อ datatype ของ PL/SQL บน Oracle 10g ตรวจจุดพังได้จาก SELECT line, position, text FROM user_errors WHERE name='SP_REPORT_EMERGENCY_ALERT'; แล้วแก้ด้วย ALTER TABLE Alert_Logs RENAME COLUMN timestamp TO alert_time; จึง compile ผ่าน — บทเรียน: อย่าตั้งชื่อคอลัมน์ตรงกับชื่อ datatype (DATE, TIMESTAMP, NUMBER, LEVEL ฯลฯ)
จุดซักถาม: ทำไม passenger_count ยังเก็บคอลัมน์ทั้งที่เป็น Derived Attribute (วงรีเส้นประใน ER)? — เก็บเป็นค่า cache ที่เปลี่ยนผ่าน Procedure เท่านั้น ไม่ใช่ข้อมูลดิบ ประโยชน์คือ query สถานะรถเร็วมาก (อ่านคอลัมน์เดียว) เทียบเท่า denormalization เพื่อ performance

-- ==========================================
-- 2. CREATE TABLES
-- ==========================================

-- 1. ตารางประเภทผู้ใช้งาน (User_Types)
CREATE TABLE User_Types (
    user_type_id VARCHAR2(10) PRIMARY KEY,
    type_name    VARCHAR2(100) NOT NULL,
    description  VARCHAR2(255)
);

-- 2. ตารางผู้ใช้งานระบบ (Users)
CREATE TABLE Users (
    user_id      VARCHAR2(20) PRIMARY KEY,
    first_name   VARCHAR2(100) NOT NULL,
    last_name    VARCHAR2(100) NOT NULL,
    phone_number VARCHAR2(20),
    email        VARCHAR2(100),
    user_type_id VARCHAR2(10),
    status       VARCHAR2(20) DEFAULT 'Active',
    CONSTRAINT fk_users_usertype FOREIGN KEY (user_type_id) REFERENCES User_Types(user_type_id)
);

-- 3. ตารางเส้นทางเดินรถ (Routes)
CREATE TABLE Routes (
    route_id    VARCHAR2(20) PRIMARY KEY,
    route_name  VARCHAR2(100) NOT NULL,
    description VARCHAR2(255),
    start_time  VARCHAR2(8),
    end_time    VARCHAR2(8),
    status      VARCHAR2(20) DEFAULT 'Active'
);

-- 4. ตารางจุดจอดรถ (Bus_Stops)
CREATE TABLE Bus_Stops (
    stop_id     VARCHAR2(20) PRIMARY KEY,
    stop_name   VARCHAR2(100) NOT NULL,
    latitude    NUMBER(10, 6),
    longitude   NUMBER(10, 6),
    description VARCHAR2(255)
);

-- 5. ตารางรายการจุดจอดในเส้นทาง (Route_Stops)
CREATE TABLE Route_Stops (
    route_id            VARCHAR2(20),
    seq                 NUMBER,
    stop_id             VARCHAR2(20),
    est_time_offset_min NUMBER,
    note                VARCHAR2(255),
    PRIMARY KEY (route_id, seq),
    CONSTRAINT fk_rs_route FOREIGN KEY (route_id) REFERENCES Routes(route_id),
    CONSTRAINT fk_rs_stop FOREIGN KEY (stop_id) REFERENCES Bus_Stops(stop_id)
);

-- 6. ตารางกล่อง IoT (IoT_Boxes)
CREATE TABLE IoT_Boxes (
    iot_id           VARCHAR2(20) PRIMARY KEY,
    latitude         NUMBER(10, 6),
    longitude        NUMBER(10, 6),
    firmware_version VARCHAR2(50),
    network_type     VARCHAR2(50),
    status           VARCHAR2(20) DEFAULT 'Active'
);

-- 7. ตารางยานพาหนะ (Vehicles)
CREATE TABLE Vehicles (
    vehicle_id      VARCHAR2(20) PRIMARY KEY,
    license_plate   VARCHAR2(20) NOT NULL,
    vehicle_model   VARCHAR2(100),
    capacity        NUMBER,
    passenger_count NUMBER DEFAULT 0,
    route_id        VARCHAR2(20),
    iot_id          VARCHAR2(20) UNIQUE,
    driver_user_id  VARCHAR2(20),
    status          VARCHAR2(20) DEFAULT 'Active',
    CONSTRAINT fk_v_route FOREIGN KEY (route_id) REFERENCES Routes(route_id),
    CONSTRAINT fk_v_iot FOREIGN KEY (iot_id) REFERENCES IoT_Boxes(iot_id),
    CONSTRAINT fk_v_driver FOREIGN KEY (driver_user_id) REFERENCES Users(user_id)
);

-- 8. ตารางปุ่มกดแจ้งเตือนฉุกเฉิน (Emergency_Buttons)
CREATE TABLE Emergency_Buttons (
    button_id       VARCHAR2(20) PRIMARY KEY,
    vehicle_id      VARCHAR2(20),
    button_location VARCHAR2(100),
    status          VARCHAR2(20) DEFAULT 'Active',
    CONSTRAINT fk_eb_vehicle FOREIGN KEY (vehicle_id) REFERENCES Vehicles(vehicle_id)
);

-- 9. ตารางการกดกริ่งดิจิตอล (Digital_Bells)
CREATE TABLE Digital_Bells (
    bell_id     VARCHAR2(20) PRIMARY KEY,
    date_time   TIMESTAMP,
    bell_status VARCHAR2(50),
    user_id     VARCHAR2(20),
    vehicle_id  VARCHAR2(20),
    note        VARCHAR2(255),
    CONSTRAINT fk_db_user FOREIGN KEY (user_id) REFERENCES Users(user_id),
    CONSTRAINT fk_db_vehicle FOREIGN KEY (vehicle_id) REFERENCES Vehicles(vehicle_id)
);

-- 10. ตารางบันทึกการแจ้งเตือนฉุกเฉิน (Alert_Logs)
CREATE TABLE Alert_Logs (
    log_id     VARCHAR2(20) PRIMARY KEY,
    vehicle_id VARCHAR2(20),
    button_id  VARCHAR2(20),
    alert_time TIMESTAMP,
    status     VARCHAR2(20),
    note       VARCHAR2(255),
    CONSTRAINT fk_al_vehicle FOREIGN KEY (vehicle_id) REFERENCES Vehicles(vehicle_id),
    CONSTRAINT fk_al_button FOREIGN KEY (button_id) REFERENCES Emergency_Buttons(button_id)
);

CREATE OR REPLACE PROCEDURE sp_press_digital_bell (
    p_user_id    IN Digital_Bells.user_id%TYPE,
    p_vehicle_id IN Digital_Bells.vehicle_id%TYPE,
    o_bell_id    OUT VARCHAR2,
    o_status     OUT VARCHAR2
) IS
    v_user_type_id   Users.user_type_id%TYPE;
    v_user_status    Users.status%TYPE;
    v_vehicle_status Vehicles.status%TYPE;
    v_next_num       NUMBER;
BEGIN
    -- ตรวจสอบผู้ใช้งานว่ามีอยู่จริงและใช้งานได้
    SELECT user_type_id, status
    INTO   v_user_type_id, v_user_status
    FROM   Users
    WHERE  user_id = p_user_id;

    IF v_user_status <> 'Active' THEN
        o_status := 'ERROR: บัญชีผู้ใช้ถูกระงับ ไม่สามารถกดกริ่งได้';
        RETURN;
    END IF;

    -- Business Rule: กดกริ่งได้เฉพาะผู้รับบริการ (ผู้โดยสาร)
    IF v_user_type_id <> 'UT-003' THEN
        o_status := 'ERROR: เฉพาะผู้โดยสาร (ประเภทผู้รับบริการ) เท่านั้นที่กดกริ่งได้';
        RETURN;
    END IF;

    -- ตรวจสอบสถานะยานพาหนะ
    SELECT status
    INTO   v_vehicle_status
    FROM   Vehicles
    WHERE  vehicle_id = p_vehicle_id;

    IF v_vehicle_status <> 'Active' THEN
        o_status := 'ERROR: ยานพาหนะไม่พร้อมให้บริการ';
        RETURN;
    END IF;

    -- สร้างรหัสกริ่งอัตโนมัติ BELL-XXX (ต่อจากเลขมากที่สุดเดิม)
    SELECT NVL(MAX(TO_NUMBER(SUBSTR(bell_id, 6))), 0) + 1
    INTO   v_next_num
    FROM   Digital_Bells
    WHERE  REGEXP_LIKE(bell_id, '^BELL-[0-9]+$');

    o_bell_id := 'BELL-' || LPAD(TO_CHAR(v_next_num), 3, '0');

    -- บันทึกเหตุการณ์กดกริ่ง
    INSERT INTO Digital_Bells (bell_id, date_time, bell_status, user_id, vehicle_id)
    VALUES (o_bell_id, SYSTIMESTAMP, 'รอดำเนินการ', p_user_id, p_vehicle_id);

    COMMIT;
    o_status := 'SUCCESS: กดกริ่งสำเร็จ รหัส ' || o_bell_id;

EXCEPTION
    WHEN NO_DATA_FOUND THEN
        ROLLBACK;
        o_status := 'ERROR: ไม่พบผู้ใช้งานหรือยานพาหนะรหัสนี้ในระบบ';
    WHEN OTHERS THEN
        ROLLBACK;
        o_status := 'ERROR: ' || SQLERRM;
END sp_press_digital_bell;

CREATE OR REPLACE PROCEDURE sp_report_emergency_alert (

    o_status    OUT VARCHAR2

    v_btn_status Emergency_Buttons.status%TYPE;
    v_next_num   NUMBER;

CREATE OR REPLACE PROCEDURE sp_update_passenger_count (
    p_vehicle_id IN Vehicles.vehicle_id%TYPE,
    p_delta      IN NUMBER,
    o_status     OUT VARCHAR2
) IS
    v_current NUMBER;
BEGIN
    -- ล็อกแถวรถและอ่านจำนวนคนปัจจุบัน
    SELECT passenger_count
    INTO   v_current
    FROM   Vehicles
    WHERE  vehicle_id = p_vehicle_id
    FOR UPDATE;

    IF v_current + p_delta < 0 THEN
        ROLLBACK;
        o_status := 'ERROR: จำนวนคนบนรถห้ามติดลบ (ปัจจุบัน = ' || v_current || ' คน)';
    ELSE
        UPDATE Vehicles
        SET    passenger_count = passenger_count + p_delta
        WHERE  vehicle_id = p_vehicle_id;

        COMMIT;
        o_status := 'SUCCESS: จำนวนคนบนรถ ' || p_vehicle_id || ' ตอนนี้ = '
                    || (v_current + p_delta) || ' คน';
    END IF;

EXCEPTION
    WHEN NO_DATA_FOUND THEN
        ROLLBACK;
        o_status := 'ERROR: ไม่พบยานพาหนะรหัสนี้';
    WHEN OTHERS THEN
        ROLLBACK;
        o_status := 'ERROR: ' || SQLERRM;
END sp_update_passenger_count;

โจทย์ข้อ 4 ของอาจารย์ — รายงานซับซ้อนที่ JOIN Transaction tables กับ Master tables มาก ๆ ครบ 5 ฉบับ ไล่จากง่ายไปยาก (5 → 6 → GROUP BY → WITH → 7 ตาราง)

อธิบายทีละรายงาน

  1. รายงาน 1 (JOIN 5 ตาราง) — ประวัติกดกริ่ง: ใครกด (Users + User_Types) ขอลงรถคันไหน (Vehicles) วิ่งสายอะไร (Routes) — INNER JOIN ทั้งหมดเพราะข้อมูลกริ่งต้องมีครบทุกส่วนอยู่แล้วตาม FK
  2. รายงาน 2 (JOIN 6 ตาราง) — เหตุฉุกเฉิน: เพิ่ม Emergency_Buttons + ใช้ LEFT JOIN กับ Users (คนขับ) และ IoT_Boxes เพราะรถอาจยังไม่มีคนขับประจำ/ไม่มีกล่อง — รายงานต้องไม่ทิ้งแถวเหตุฉุกเฉินแม้ข้อมูลรถไม่สมบูรณ์
  3. รายงาน 3 (JOIN 4 + GROUP BY + Analytic) — ภาพรวมสาย: จำนวนรถ จำนวนป้าย ป้ายแรก-ป้ายสุดท้าย ใช้ MIN/MAX(...) KEEP (DENSE_RANK FIRST/LAST ORDER BY rs.seq) — analytic function เลือกค่าจากแถว seq น้อยสุด/มากสุดโดยไม่ต้อง subquery ซ้อน ⚠️ นับจำนวนป้ายต้องใช้ COUNT(DISTINCT rs.seq) ไม่ใช่ COUNT(rs.seq) เพราะ LEFT JOIN กับ Vehicles ทำให้ 1 ป้ายถูกนับซ้ำเท่าจำนวนรถในสาย (fan-out) — บั๊กจริงที่จับได้วันนี้: RT-01 โชว์ 10 ป้ายทั้งที่จริง 5
  4. รายงาน 4 (WITH clause) — สถิติเหตุการณ์ต่อสาย: รวมกริ่ง (Bell_Stats) กับฉุกเฉิน (Alert_Stats) แยกกันก่อน 2 CTE แล้วจึง JOIN เข้า Routes — เหตุผล: (1) กัน fan-out ข้อมูลซ้ำจากการ JOIN ตรงหลายตาราง (2) NVL ครอบค่า 0 สำหรับสายที่ไม่มีเหตุการณ์ (3) ใช้ WITH แทน CREATE VIEW เพราะ user ไม่มีสิทธิ์ CREATE VIEW (ORA-01031)
  5. รายงาน 5 (JOIN 7 ตาราง — มากที่สุด) — เหตุกริ่งแต่ละครั้ง + จุดจอดทั้งสายของคันนั้น: Digital_Bells → Users → User_Types → Vehicles → Routes → Route_Stops → Bus_Stops — แสดงว่ากริ่งเกิดบนสายอะไร และรถต้องแวะป้ายไหนตามลำดับ seq

-- รายงานที่ 1: ประวัติการกดกริ่งดิจิตอลขอลงรถ (JOIN 5 ตาราง)

    r.route_name

JOIN Users      u  ON db.user_id     = u.user_id
JOIN User_Types ut ON u.user_type_id = ut.user_type_id

JOIN Emergency_Buttons eb ON al.button_id      = eb.button_id
JOIN Vehicles   v         ON al.vehicle_id     = v.vehicle_id
JOIN Routes     r         ON v.route_id        = r.route_id

-- ============================================================
-- รายงานที่ 3: สรุปภาพรวมแต่ละสายเดินรถ (JOIN 4 ตาราง + GROUP BY)
-- ============================================================
SELECT
    r.route_id,
    r.route_name,
    COUNT(DISTINCT v.vehicle_id)                              AS total_vehicles,
    COUNT(DISTINCT rs.seq)                                    AS total_stops,
    MIN(bs.stop_name) KEEP (DENSE_RANK FIRST ORDER BY rs.seq) AS first_stop,
    MAX(bs.stop_name) KEEP (DENSE_RANK LAST  ORDER BY rs.seq) AS last_stop
FROM Routes r
LEFT JOIN Vehicles    v  ON r.route_id = v.route_id
LEFT JOIN Route_Stops rs ON r.route_id = rs.route_id
LEFT JOIN Bus_Stops   bs ON rs.stop_id = bs.stop_id
GROUP BY r.route_id, r.route_name
ORDER BY r.route_id;

-- ============================================================
-- รายงานที่ 4: สรุปจำนวนเหตุการณ์แยกตามสาย (WITH clause + JOIN)
-- ============================================================
WITH Bell_Stats AS (
    SELECT v.route_id, COUNT(*) AS bell_count
    FROM Digital_Bells db
    JOIN Vehicles v ON db.vehicle_id = v.vehicle_id
    GROUP BY v.route_id
),
Alert_Stats AS (
    SELECT v.route_id, COUNT(*) AS alert_count
    FROM Alert_Logs al
    JOIN Vehicles v ON al.vehicle_id = v.vehicle_id
    GROUP BY v.route_id
)
SELECT
    r.route_id,
    r.route_name,
    NVL(bs.bell_count, 0)   AS total_bell_presses,
    NVL(ast.alert_count, 0) AS total_emergency_alerts
FROM Routes r
LEFT JOIN Bell_Stats  bs  ON r.route_id = bs.route_id
LEFT JOIN Alert_Stats ast ON r.route_id = ast.route_id
ORDER BY r.route_id;

-- รายงานที่ 5: เหตุการณ์กดกริ่ง + จุดจอดทั้งหมดในเส้นทางของคันนั้น
-- (JOIN มากที่สุด 7 ตาราง)
-- ============================================================

  1. Query หลักของอาจารย์ (แก้บั๊ก) — ต้นฉบับใช้ USER_TABLES.OWNER ซึ่งไม่มีอยู่จริง (ORA-00904) แก้เป็น USER AS OWNER — LEFT OUTER JOIN เพื่อให้ตารางที่ไม่มี constraint ก็โผล่ (ไม่หายจากรายงาน)
  2. Query เสริม constraint รายคอลัมน์ — USER_CONSTRAINTS + USER_CONS_COLUMNS แสดงว่า constraint แต่ละตัวอยู่ที่คอลัมน์ไหน ลำดับไหน (POSITION สำคัญกับ PK/FK composite เช่น (ROUTE_ID, SEQ))
  3. Query กรองเฉพาะงาน SE — WHERE table_name IN (...10 ตาราง...) เพราะ user มีตาราง BA ปนอยู่ (รวม 18 ตาราง)
จำ CONSTRAINT_TYPE: P = Primary Key · R = Foreign Key · U = Unique · C = Check/NOT NULL · SYS_Cxxxx = Oracle ตั้งชื่อให้เอง
สคริปต์เต็ม 
-- ============================================================
-- Script ตรวจสอบผลการทำงาน: แสดง TABLE ที่ User เป็นเจ้าของ + Constraint
-- (ของอาจารย์ — เวอร์ชันแก้ไขให้รันได้จริง)
-- ============================================================
SELECT
    t.TABLE_NAME,
    t.TABLESPACE_NAME,
    USER AS OWNER,
    c.CONSTRAINT_NAME,
    c.CONSTRAINT_TYPE,
    c.TABLE_NAME
FROM USER_TABLES t
LEFT OUTER JOIN USER_CONSTRAINTS c
       ON (t.TABLE_NAME = c.TABLE_NAME)
ORDER BY t.TABLE_NAME, c.CONSTRAINT_TYPE, c.CONSTRAINT_NAME;

-- ============================================================
-- (เสริม) แสดงคอลัมน์ที่เกี่ยวข้องกับแต่ละ Constraint
-- ============================================================
SELECT
    c.TABLE_NAME,
    c.CONSTRAINT_NAME,
    c.CONSTRAINT_TYPE,
    cc.COLUMN_NAME,
    cc.POSITION
FROM USER_CONSTRAINTS c
LEFT JOIN USER_CONS_COLUMNS cc
       ON c.CONSTRAINT_NAME = cc.CONSTRAINT_NAME
      AND c.OWNER = cc.OWNER
ORDER BY c.TABLE_NAME, c.CONSTRAINT_TYPE, c.CONSTRAINT_NAME, cc.POSITION;

-- ============================================================
-- (เสริม) กรองเฉพาะตารางของโปรเจกต์นี้ (Smart Transit)
-- ============================================================
SELECT
    t.TABLE_NAME,
    t.TABLESPACE_NAME,
    USER AS OWNER,
    c.CONSTRAINT_NAME,
    c.CONSTRAINT_TYPE,
    c.TABLE_NAME
FROM USER_TABLES t
LEFT OUTER JOIN USER_CONSTRAINTS c
       ON (t.TABLE_NAME = c.TABLE_NAME)
WHERE t.TABLE_NAME IN ('USER_TYPES','USERS','ROUTES','BUS_STOPS','ROUTE_STOPS',
                       'IOT_BOXES','VEHICLES','EMERGENCY_BUTTONS',
                       'DIGITAL_BELLS','ALERT_LOGS')
ORDER BY t.TABLE_NAME, c.CONSTRAINT_TYPE, c.CONSTRAINT_NAME;
